FORM get_employee USING    p_uname
                  CHANGING p_pernr
                           p_pernr_name.

  CLEAR: p_pernr, p_pernr_name.

  SELECT SINGLE * FROM  pa0105
         WHERE  usrty  = '0001'
         AND    endda  GE sy-datum
*        AND    begda  LE sy-datum
         AND    usrid  = p_uname.

  IF  sy-subrc NE 0.

  ELSE.
    p_pernr = pa0105-pernr.
  ENDIF.

  PERFORM get_employee_name USING    p_pernr

FORM gui_indicator  USING    p_count
                             p_lines.

  DATA: l_per          TYPE i,
         l_text(50)     TYPE c.

  l_per = p_count * 100 / p_lines.
  WRITE l_per TO l_text NO-ZERO.
  SHIFT l_text LEFT DELETING LEADING ' '.
  CONCATENATE l_text '% Processed' INTO l_text.
  CALL FUNCTION 'SAPGUI_PROGRESS_INDICATOR'
    EXPORTING
      percentage = l_per
      text       = l_text.


ENDFORM.                    " GUI_INDICATOR

*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*
*  -->  LINES                                                         *
*  -->  HEADER                                                        *
*---------------------------------------------------------------------*
FORM add_user_line TABLES lines STRUCTURE tline
                   USING p_length.

  DATA: tan_date(10)  TYPE c,
        tan_time(8)   TYPE c,
        tan_uname     TYPE p1000-stext,
        tan_realo     TYPE hrealo-realo,
        l_lines       TYPE i,
        l_length      TYPE i.

  DESCRIBE TABLE lines LINES l_lines.
  IF l_lines NE 0.
    CLEAR lines.
    APPEND lines.
  ENDIF.

*--- prepare variables -------------------------------------------------
  WRITE sy-datlo TO tan_date DD/MM/YYYY.
  WRITE sy-timlo TO tan_time.
  MOVE sy-uname TO tan_realo.
  CALL FUNCTION 'RH_READ_OBJECT'
    EXPORTING
      plvar           = '!!'
      otype           = 'US'
      realo           = tan_realo
      check_stru_auth = ' '
    IMPORTING
      stext           = tan_uname
    EXCEPTIONS
      not_found       = 1
      OTHERS          = 2.
  IF sy-subrc <> 0.
    CLEAR tan_uname.
  ENDIF.

*--- compose first line ------------------------------------------------
  CLEAR lines.
  CONCATENATE 'Note created by' tan_uname '('
              INTO lines-tdline SEPARATED BY ' '.
  CONCATENATE lines-tdline sy-uname ')'
              INTO lines-tdline.
  CONCATENATE lines-tdline 'on' tan_date 'at' tan_time
              INTO lines-tdline SEPARATED BY ' '.
  l_length = strlen( lines-tdline ).
  IF l_length GT p_length.
    CONCATENATE 'Note created by' tan_uname '('
                INTO lines-tdline SEPARATED BY ' '.
    CONCATENATE lines-tdline sy-uname ')'
                INTO lines-tdline.
    APPEND lines.
    CONCATENATE 'Note created on' tan_date 'at' tan_time
                INTO lines-tdline SEPARATED BY ' '.
    APPEND lines.
  ELSE.
    APPEND lines.
  ENDIF.

ENDFORM.                    " ADD_USER_LINE

*      -->P_GR_OUTTAB4_PERNR  text
*--------------------------------------------------------------------*
FORM get_emp_orgeh  USING    p_pernr
                    CHANGING p_orgeh.

  CLEAR p_orgeh.

  SELECT SINGLE * FROM  pa0001
         WHERE  pernr  EQ p_pernr
           AND    endda  GE sy-datum
           AND    begda  LE sy-datum.

  IF  sy-subrc NE 0.
    CLEAR pa0001.
  ENDIF.

* determine if there is a current hda
  SELECT SINGLE * FROM  pa0509
       WHERE  pernr  EQ p_pernr
       AND    sprps  EQ ' '
       AND    endda  GE sy-datum
       AND    begda  LE sy-datum.
  IF sy-subrc = 0.
* determine the hda org unit
    SELECT SINGLE * FROM hrp1001
            WHERE otype =  'S'
            AND   objid =  pa0509-plans
            AND   plvar =  c_plvar_01
            AND   rsign =  'A'
            AND   relat =  '003'
            AND   istat =  '1'
            AND   priox LIKE '%'
            AND begda <= sy-datum
            AND endda >= sy-datum.
    IF sy-subrc = 0.
      p_orgeh = hrp1001-sobid(8).
    ENDIF.
  ENDIF.

  IF p_orgeh IS INITIAL.
    p_orgeh = pa0001-orgeh.
  ENDIF.


ENDFORM.                    " GET_EMP_ORGEH

*--------------------------------------------------------------------*
*       Form  SAVE_TEXT_IT9203
*--------------------------------------------------------------------*
*       text
*--------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*--------------------------------------------------------------------*
FORM save_text_it9203 TABLES tb_outtxt    STRUCTURE gr_outtxt1
                       USING  p_code
                              p_seq.

  DATA:  lt_longs          TYPE table_of_strings,
          l_longt           TYPE char255,
          l_longs           TYPE string,
          index(3)          TYPE n,
          act_index         TYPE sytabix,
          length            TYPE sytabix,
          maxlen            TYPE sytabix,
          finish(1)         TYPE c,
          num(2)            TYPE n,
          l_cr_lf(2)        TYPE c,
          stream_text       TYPE string.

  DATA: lt_longt           TYPE char255 OCCURS 0 WITH HEADER LINE.
  DATA: tb_outtxt2         TYPE gs_outtxt_type2
                                 OCCURS 0 WITH HEADER LINE.

  tb_outtxt2[] = tb_outtxt[].

  CALL FUNCTION 'CONVERT_TABLE_TO_STRING'
    EXPORTING
      i_tabline_length = 100
    IMPORTING
      e_string         = stream_text
    TABLES
      it_table         = tb_outtxt2.

  CLEAR   : lt_longs, lt_longt.
  REFRESH : lt_longs, lt_longt.
  l_cr_lf = cl_abap_char_utilities=>cr_lf.
  SPLIT  stream_text AT  l_cr_lf INTO TABLE  lt_longs .
  LOOP AT lt_longs INTO l_longs.
    length = strlen( l_longs ).
    IF length <= 75.
      l_longt = l_longs.
      APPEND l_longt TO lt_longt.
    ELSE.
      act_index = 0.
      WHILE act_index < length.
        maxlen = length - act_index.
        IF maxlen > 75.
          maxlen = 75.
        ENDIF.
        l_longt = l_longs+act_index(maxlen).
        CLEAR finish.
        index = 74.
        WHILE finish  <> abap_true AND index <> 0.
          IF l_longt+index(1) = ' '.
            finish = abap_true.
          ENDIF.
          index = index - 1.
        ENDWHILE.
        index = index + 1.
        IF index <> 0.
          l_longt = l_longt(index).
          APPEND l_longt TO lt_longt.
          act_index = act_index + index + 1.
        ELSE.
          APPEND l_longt TO lt_longt.
          act_index = act_index + 75.
        ENDIF.
      ENDWHILE.
    ENDIF.
  ENDLOOP.


  CLEAR w_thead.
  w_thead-tdobject = 'ZPOLICE'.
  CONCATENATE zim_in_mast-pernr p_seq p_code
              INTO w_text_key.
  w_thead-tdname   = w_text_key.
  w_thead-tdid     = 'ZIM3'.
  w_thead-tdform   = 'SYSTEM'.
  w_thead-tdstyle  = 'HR_TEM1'.
  w_thead-tdspras  = 'E'.


  REFRESH tbl_tline.
  CLEAR tbl_tline.

  LOOP AT lt_longt.
    tbl_tline-tdline = lt_longt.
    APPEND tbl_tline.
  ENDLOOP.

  CALL FUNCTION 'SAVE_TEXT'
    EXPORTING
      header          = w_thead
      savemode_direct = abap_true
    TABLES
      lines           = tbl_tline
    EXCEPTIONS
      id              = 1
      language        = 2
      name            = 3
      object          = 4
      OTHERS          = 5.

ENDFORM.                    " SAVE_TEXT_IT9203

  DATA: l_subrc     TYPE sy-subrc,
        l_tabix     TYPE sy-tabix,
        l_key(19)   TYPE c,
        l_uname(12) TYPE c,
        l_ename(40) TYPE c.

  DATA: BEGIN OF l_enq OCCURS 100.
          INCLUDE STRUCTURE seqg3.
  DATA: END OF l_enq.

  DATA: l_pernr         TYPE zim_in_mast-pernr.

  l_pernr = p_pernr.


  CALL FUNCTION 'ENQUEUE_EZIM_IN_MAST'
    EXPORTING
      mode_zim_in_mast = 'E'
      mandt            = sy-mandt
      pernr            = l_pernr
      x_pernr          = ' '
      _scope           = '2'
      _wait            = ' '
      _collect         = ' '
    EXCEPTIONS
      foreign_lock     = 1
      system_failure   = 2
      OTHERS           = 3.
  IF sy-subrc <> 0.
    PERFORM unlock_im_pernr USING l_pernr.

    CLEAR:   l_tabix, l_subrc, l_enq, l_key, l_uname.
    REFRESH: l_enq.

*    CALL FUNCTION 'ENQUE_READ'     "DEVK9HR5A4
    CALL FUNCTION 'ENQUEUE_READ'                            "DEVK9HR5A4
      EXPORTING
        gclient = sy-mandt
        gname   = ' '
        garg    = ' '
        guname  = ' '
      IMPORTING
        number  = l_tabix
        subrc   = l_subrc
      TABLES
        enq     = l_enq.

    IF sy-subrc EQ 0.
      CONCATENATE sy-mandt p_pernr
        INTO l_key.

      LOOP AT l_enq
        WHERE gname = 'ZIM_IN_MAST'
        AND   gobj  = 'EZIM_IN_MAST'
        AND   garg(11)  = l_key.
        l_uname = l_enq-guname.
        EXIT.
      ENDLOOP.

      PERFORM get_user_name  USING    l_uname
                             CHANGING l_ename.

      MESSAGE e999 WITH  'Injury Management Case for'
                         p_pernr
                         ' is currently locked by '
                         l_ename.
    ENDIF.
  ENDIF.

FORM retrieve_pa9001_text .

  DATA: test              TYPE zhr_inc_tbody,
        fieldname(30)     TYPE c,
        fieldname1(30)    TYPE c,
        lt_longs          TYPE table_of_strings,
        l_longt           TYPE char255,
        l_longs           TYPE string,
        index(3)          TYPE n,
        act_index         TYPE sytabix,
        length            TYPE sytabix,
        maxlen            TYPE sytabix,
        finish(1)         TYPE c,
        num(2)            TYPE n,
        l_cr_lf(2)        TYPE c,
        l_struc(10)       TYPE c,
        pkey              TYPE pskey,
        l_space(3)        TYPE c,
        l_space_new(3)    TYPE c,
        tbl_string        TYPE string_table,
        rec_string        TYPE string,
        l_lines           TYPE i.

  tx-key-infty = '9001'.

  CLEAR stream_text.
  REFRESH ptext. CLEAR ptext.

  IMPORT pkey TO pkey ptext TO ptext
  FROM DATABASE pcl1(tx)
  ID tx-key.
  CALL FUNCTION 'CONVERT_TABLE_TO_STRING'
    EXPORTING
      i_tabline_length = 72
    IMPORTING
      e_string         = stream_text
    TABLES
      it_table         = ptext.

* Under some situations the CRLF is split over two lines and when
* the lines are merged a space is inserted between the CR and the LF.

  l_cr_lf = cl_abap_char_utilities=>cr_lf.

  CONCATENATE l_cr_lf(1) l_cr_lf+1(1)
              INTO l_space SEPARATED BY ' '.

  CONCATENATE l_cr_lf(1) l_cr_lf+1(1)
              INTO l_space_new.

  REPLACE ALL OCCURRENCES OF l_space IN stream_text
          WITH l_space_new.

* Remove any spaces at the beginning of the line
  SPLIT stream_text AT l_space_new
        INTO TABLE tbl_string.
  DESCRIBE TABLE tbl_string LINES l_lines.
  IF l_lines GT 1.
    CLEAR stream_text.
    LOOP AT tbl_string INTO rec_string.
      IF stream_text IS NOT INITIAL.
        CONCATENATE stream_text rec_string
                    INTO stream_text
                    SEPARATED BY l_cr_lf.
      ELSE.
        stream_text = rec_string.
      ENDIF.
    ENDLOOP.
  ENDIF.


  SPLIT  stream_text AT  l_cr_lf INTO TABLE  lt_longs .
  LOOP AT lt_longs INTO l_longs.
    length = strlen( l_longs ).
    IF length <= 65.
      l_longt = l_longs.
      APPEND l_longt TO lt_longt.
    ELSE.
      act_index = 0.
      WHILE act_index < length.
        maxlen = length - act_index.
        IF maxlen > 65.
          maxlen = 65.
        ENDIF.
        l_longt = l_longs+act_index(maxlen).
        CLEAR finish.
        index = 64.
        WHILE finish  <> abap_true AND index <> 0.
          IF l_longt+index(1) = ' '.
            finish = abap_true.
          ENDIF.
          index = index - 1.
        ENDWHILE.
        index = index + 1.
        IF index <> 0.
          l_longt = l_longt(index).
          APPEND l_longt TO lt_longt.
          act_index = act_index + index + 1.
        ELSE.
          APPEND l_longt TO lt_longt.
          act_index = act_index + 65.
        ENDIF.
      ENDWHILE.
    ENDIF.
  ENDLOOP.

ENDFORM.                    " RETRIEVE_PA9001_TEXT

FORM send_email_p902 .

  DATA: l_text(80)        TYPE c,
         l_count(2)        TYPE n,
         l_fieldname(8)    TYPE c,
         l_name(120)       TYPE c.

  FIELD-SYMBOLS : <fs>.

  WRITE wa_incident-pernr TO l_name LEFT-JUSTIFIED.
  SHIFT l_name LEFT DELETING LEADING '0'.
  CONCATENATE '(' l_name ')' INTO l_name.
  CONCATENATE wa_incident-titel wa_incident-vorna wa_incident-nachn
              l_name INTO l_name SEPARATED BY ' '.
  REFRESH : tbl_email.
  CLEAR   : tbl_email.
  l_count = 1.
  WHILE l_count LE 12.
    CONCATENATE 'TEXT-T' l_count INTO l_fieldname.
    ASSIGN (l_fieldname) TO <fs>.
    tbl_email-rec = <fs>.
    REPLACE '&1' IN tbl_email-rec WITH l_name.
    APPEND tbl_email.
    ADD 1 TO l_count.
  ENDWHILE.

  tbl_objbin[] = tbl_pdf[].
  CONCATENATE '(' wa_incident-identnr ')' INTO l_name.
  CONCATENATE 'Incident Report' l_name
              wa_incident-vorna wa_incident-nachn
              INTO w_title SEPARATED BY ' '.
  WRITE wa_incident-pernr TO l_name LEFT-JUSTIFIED.
  SHIFT l_name LEFT DELETING LEADING '0'.
  CONCATENATE 'Servicenr' l_name
              INTO w_title_att SEPARATED BY ' '.

  IF wa_incident-pre88 = abap_true.
    SELECT SINGLE * FROM zhr_inc_mailrec
           WHERE id = 'ALIA1'.
  ELSE.
    SELECT SINGLE * FROM zhr_inc_mailrec
           WHERE id = 'EML'.
  ENDIF.

  PERFORM send_email USING zhr_inc_mailrec-email_adr.


ENDFORM.                    " SEND_EMAIL_P902

*--------------------------------------------------------------------*
FORM remove_path  CHANGING p_filename.

  DATA: l_filename         TYPE zim_in_filename,
         lv_str_len         TYPE i,
         lv_str_file        TYPE string,
         lv_chr_ext(3)      TYPE c,
         i                  TYPE i,
         j                  TYPE i,
         k(1)               TYPE c.

  lv_str_file = p_filename.
  lv_str_len = strlen( lv_str_file ).
  lv_str_len = lv_str_len - 3.
  lv_chr_ext = lv_str_file+lv_str_len(3).

* Remove the path from the filename
  i = 0.
  lv_str_len = lv_str_len + 3.
  j = 0.
  CLEAR l_filename.
  WHILE i LT lv_str_len.
    k = p_filename+i(1).
    IF k = '\'.
      CLEAR: l_filename, j.
    ELSE.
      l_filename+j(1) = k.
      ADD 1 TO j.
    ENDIF.
    ADD 1 TO i.
  ENDWHILE.

  p_filename = l_filename.

ENDFORM.                    " REMOVE_PATH

FORM read_it_text USING p_pernr
                        p_infotype
                        p_subty
                        p_begda
                        p_endda
                        p_seqnr.

  DATA: tx-key        TYPE pskey.
  DATA: p_line(78) TYPE c.

  tx-key-pernr = p_pernr.
  tx-key-infty = p_infotype.
  tx-key-subty = p_subty.
  tx-key-begda = p_begda.
  tx-key-endda = p_endda.
  tx-key-seqnr = p_seqnr.
  rp-imp-c1-tx.

  CLEAR: w_text_begda, w_text_endda, w_text_incident.
  DESCRIBE TABLE ptext LINES w_lines.
  IF w_lines GE 1.
    READ TABLE ptext INDEX 1.
    IF p_subty = 'WCDA' OR
       p_subty = 'WCDE' OR
       p_subty = 'WCDR'.
      w_text_begda = ptext+22(10).
      w_text_endda = ptext+36(10).
    ELSE.
      w_text_begda = ptext+29(10).
      w_text_endda = ptext+43(10).
    ENDIF.
  ENDIF.
  IF w_lines GE 2.
    READ TABLE ptext INDEX 2.
    w_text_incident = ptext+16(10).
  ENDIF.


ENDFORM.                    " READ_IT_TEXT

         l_date(10)                TYPE c.

    EXPORTING
      date_in             = l_date
      date_format_in      = 'DDMY'
      to_output_format    = abap_true

*&---------------------------------------------------------------------*
*       Get telephone number
*----------------------------------------------------------------------*
FORM get_telephone_no  USING    pv_pernr  TYPE persno
                       CHANGING pv_phone  TYPE char20.

  DATA:  ls_pa0006    TYPE pa0006,
         lv_cnt       TYPE num2     VALUE '03',
         lv_fname     TYPE string.
  FIELD-SYMBOLS:
         <lv_num>     TYPE pa0006-num01.

  CLEAR pv_phone.
  CHECK pv_pernr IS NOT INITIAL.
* get telephone number
  SELECT SINGLE * FROM pa0006 INTO ls_pa0006
          WHERE pernr = pv_pernr
            AND subty = '99'
            AND begda LE sy-datum
            AND endda GE sy-datum.
  IF sy-subrc EQ 0.
    lv_fname = |NUM{ lv_cnt }|.
    ASSIGN COMPONENT lv_fname OF STRUCTURE ls_pa0006 TO <lv_num>.
    IF  sy-subrc EQ 0
    AND <lv_num> IS NOT INITIAL.
      pv_phone = <lv_num>.
      EXIT.       "Exit Do loop on first find
    ENDIF.
  ENDIF.

ENDFORM.                    " GET_TELEPHONE_NO
