*      -->P_3165   text
*--------------------------------------------------------------------*
FORM maintain_wc_benefits   USING  p_action
                                   p_begda
                                   p_endda
                                   p_seq_no.

  DATA: l_begda          TYPE zim_in_cert-date_from,
         l_endda          TYPE zim_in_cert-date_to.

* MASC Integration
* Check if the employee is currently suspended
* If they are, do not allow GAO's to process the record
  DATA: l_zmsc_sus             TYPE zmsc_sus.

  IF sy-tcode = 'ZWC_GAO'.
    SELECT SINGLE * FROM zmsc_sus
           INTO l_zmsc_sus
           WHERE pernr  = zim_in_mast-pernr
             AND status = 'S'.
    IF sy-subrc = 0.
      MESSAGE e999 WITH 'P905 records for Suspended'
                        'Officers must be processed'
                        'by Shared Services'.
    ENDIF.
  ENDIF.

  CLEAR w_code2.
  IF p_begda = '0'.
    CLEAR : l_begda, l_endda.
  ELSE.
    l_begda = p_begda.
    l_endda = p_endda.
    IF l_endda IS INITIAL.
      l_endda = '99991231'.
    ENDIF.
  ENDIF.

  CALL FUNCTION 'Z_OLIMS_WC_BENEFITS'
    EXPORTING
      i_action          = p_action
      i_pernr           = zim_in_mast-pernr
      i_incident        = zim_in_mast-incident
      i_begda           = l_begda
      i_endda           = l_endda
      i_seq_no          = p_seq_no
      i_wkly_wc_plan    = w_ip_det-wkly_wc_plan
      i_est_end_date_p1 = w_epss_det-estimated_end_date_p1
    EXCEPTIONS
      recoup_not_found  = 1
      invalid_action    = 2
      OTHERS            = 3.
  IF sy-subrc = 1.
    MESSAGE e999 WITH 'Invalid Recoup'.
  ELSEIF sy-subrc = 2.
    MESSAGE e999 WITH 'Invalid Action'.
  ENDIF.

  w_3405_refresh = abap_true.

* The user may have updated Entitlement Period Bucket dates,
* so need to invalidate the current Buckets detail forcing
* it to be recalculated when displayed.
  CLEAR w_epb_det.

ENDFORM.                    " MAINTAIN_WC_BENEFITS

*--------------------------------------------------------------------*
*      -->P_P_PERNR  text
*      -->P_P_INCIDENT  text
*--------------------------------------------------------------------*
FORM get_health_details  USING    p_pernr
                                  p_incident.

  DATA: l_health_prof         TYPE zim_med-health_prof.

  REFRESH gt_outtab6.
  CLEAR : w_ntd_name, w_ntd_phone, w_nts_name, w_nts_phone.

  IF zim_in_mast-ntd IS NOT INITIAL.
    SELECT SINGLE * FROM zim_med
           WHERE med_prac = zim_in_mast-ntd.
    IF sy-subrc = 0.
      w_ntd_name  = zim_med-name.
      w_ntd_phone = zim_med-phone.
    ENDIF.
  ENDIF.

  IF zim_in_mast-nts IS NOT INITIAL.
    SELECT SINGLE * FROM zim_med
           WHERE med_prac = zim_in_mast-nts.
    IF sy-subrc = 0.
      w_nts_name  = zim_med-name.
      w_nts_phone = zim_med-phone.
    ENDIF.
  ENDIF.

  SELECT * FROM zim_in_med
         INTO CORRESPONDING FIELDS OF TABLE gt_outtab6
         WHERE pernr    = p_pernr
           AND incident = p_incident.

  LOOP AT gt_outtab6 INTO gr_outtab6.
    SELECT SINGLE * FROM zim_med
           WHERE med_prac = gr_outtab6-med_prac.
    IF sy-subrc = 0.
      gr_outtab6-name = zim_med-name.
      gr_outtab6-phone = zim_med-phone.
      gr_outtab6-fax   = zim_med-fax.
      gr_outtab6-email = zim_med-email.
      gr_outtab6-health_prof = zim_med-health_prof.
    ENDIF.
    IF zim_in_mast-ntd = gr_outtab6-med_prac.
      gr_outtab6-ntd = abap_true.
    ENDIF.
    IF zim_in_mast-nts = gr_outtab6-med_prac.
      gr_outtab6-nts = abap_true.
    ENDIF.
    gr_outtab6-ins_icon = icon_create.
    gr_outtab6-upd_icon = icon_change.
    MODIFY gt_outtab6 FROM gr_outtab6.
  ENDLOOP.

  SORT gt_outtab6 BY health_prof.
  CLEAR l_health_prof.
  LOOP AT gt_outtab6 INTO gr_outtab6.
    IF gr_outtab6-health_prof NE l_health_prof.
      l_health_prof = gr_outtab6-health_prof.
      SELECT SINGLE * FROM zim_med_proft
             WHERE sprsl = 'E'
               AND health_prof = gr_outtab6-health_prof.
      IF sy-subrc NE 0.
        CLEAR zim_med_proft.
      ENDIF.
    ENDIF.
    gr_outtab6-health_desc = zim_med_proft-health_desc.
    MODIFY gt_outtab6 FROM gr_outtab6.
  ENDLOOP.

  SORT gt_outtab6 BY seq_no.
  CLEAR gr_outtab6.
  gr_outtab6-ins_icon = icon_create.
  gr_outtab6-upd_icon = icon_change.
  APPEND gr_outtab6 TO gt_outtab6.
  APPEND gr_outtab6 TO gt_outtab6.

  gt_outtab6_old[] = gt_outtab6[].

ENDFORM.                    " GET_HEALTH_DETAILS

         INTO zim_in_mast-inj_location

           AND location_code = l_location_code.

  ELSEIF p9001-nearmiss = '4'.

  ELSEIF p9001-nearmiss = '5'.

  ELSEIF p9001-nearmiss = '6'.
    zim_in_mast-inc_type   = 'H'.
    zim_in_mast-ass_status = 'C'.
  ENDIF.
  zim_in_mast-inc_status = 'I'.
  zim_in_mast-date_created = sy-datum.

                       CHANGING w_user_pernr
                                w_user_name.

  SELECT SINGLE * FROM pa0220
         WHERE ( subty = '9PSS' OR subty =   '9PSP'
             OR  subty = '9PSO' OR subty =   '9PSR'
             OR  subty = 'ZPSS' OR subty =   'ZPSR' )
           AND pernr = p9001-pernr
           AND begda LE sy-datum
           AND endda GE sy-datum.
  IF sy-subrc = 0.
    zim_in_mast-hod = abap_true.
  ENDIF.

  PERFORM get_emp_orgeh USING    zim_in_mast-pernr
                        CHANGING l_orgeh.
  SELECT SINGLE * FROM zim_org
         WHERE orgeh = l_orgeh.
  IF sy-subrc = 0.
    zim_in_mast-restricted = abap_true.
  ENDIF.

  SELECT SINGLE * FROM zhorgstruc
         WHERE unit_code = l_orgeh
           AND vbegda   LE sy-datum
           AND vendda   GE sy-datum.
  IF sy-subrc = 0.
    IF  zhorgstruc-lvl3_code NE 0.
      l_orgeh = zhorgstruc-lvl3_code.
    ENDIF.

                                 'O'
                        CHANGING zim_in_mast-ima_region
                                 w_short.

                                     zim_in_mast.
  ENDIF.

    PERFORM retrieve_p902_info USING 'N'.
    PERFORM send_insurer_email.
    MESSAGE i999 WITH 'The Incident'
                      p9001-identnr
                      'has been converted from a Near Miss'.
  ELSE.

                      p9001-identnr

  ENDIF.

*&      Form  SEND_TO_MDRP
*--------------------------------------------------------------------*
*       text
*--------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*--------------------------------------------------------------------*
FORM send_to_mdrp .

  DATA: l_review_date       TYPE d.

* First lock all the cases
  LOOP AT gt_outtab7 INTO gr_outtab7
       WHERE sel_code = abap_true.
    PERFORM lock_im_pernr USING gr_outtab7-pernr.
  ENDLOOP.

  LOOP AT gt_outtab7 INTO gr_outtab7
       WHERE sel_code = abap_true.

* Send an email to Metlife
    REFRESH r_cat.
    PERFORM add_r_cat USING '33'.
    w_ltr_category = '33'.
    w_doc_unique = 'FTCM'.
    w_doc_link = 'F'.
    CLEAR w_canc.
    zim_in_mast-pernr = gr_outtab7-pernr.
    PERFORM get_employee_name USING    gr_outtab7-pernr
                              CHANGING w_emp_name.
    PERFORM correspondence USING gr_outtab7-pernr
                                 gr_outtab7-incident
                                 w_doc_unique
                                 ' '.
    IF w_canc IS NOT INITIAL.
* If the email to Metlife is not successful, do not send the
* employee to MDU.
      CLEAR gr_outtab7-sel_code.
      MODIFY gt_outtab7 FROM gr_outtab7.
      CONTINUE.
    ENDIF.

    SELECT SINGLE * FROM zim_in_ftc
           WHERE pernr    = gr_outtab7-pernr
             AND incident = gr_outtab7-incident.

    zim_in_ftc-mdu_date = sy-datum.

    w_action           = 'U'.
    UPDATE zim_in_ftc.
    IF sy-subrc NE 0.
      w_subrc = sy-subrc.
      ROLLBACK WORK.
      MESSAGE e999 WITH 'Error occurred when updating ZIM_IN_FTC'
                        'Return code ' w_subrc.
    ELSE.
      PERFORM update_in_ftc_log USING w_action.
    ENDIF.

    SELECT SINGLE * FROM zim_in_md
           WHERE pernr    = gr_outtab7-pernr
             AND incident = gr_outtab7-incident.
    IF sy-subrc = 0.
      w_action           = 'U'.
    ELSE.
      w_action           = 'I'.
      CLEAR zim_in_md.
      zim_in_md-pernr    = gr_outtab7-pernr.
      zim_in_md-incident = gr_outtab7-incident.
    ENDIF.

    zim_in_md-mdrp_date = zim_screen-mdrp_date.
    zim_in_md-nom_lds   = zim_screen-mdrp_date + 14.
    zim_in_md-req_lds   = zim_screen-mdrp_date + 14.

    IF w_action = 'I'.
      INSERT zim_in_md.
    ELSE.
      UPDATE zim_in_md.
    ENDIF.
    IF sy-subrc NE 0.
      w_subrc = sy-subrc.
      ROLLBACK WORK.
      MESSAGE e999 WITH 'Error occurred when updating ZIM_IN_MD'
                        'Return code ' w_subrc.
    ELSE.
      PERFORM update_in_md_log USING w_action.
    ENDIF.

    SELECT SINGLE * FROM zim_in_mast
           WHERE pernr    = gr_outtab7-pernr
             AND incident = gr_outtab7-incident.

    zim_in_mast-last_act_date = sy-datum.
    zim_in_mast-inc_status    = 'M'.

    w_action           = 'U'.
    UPDATE zim_in_mast.
    IF sy-subrc NE 0.
      w_subrc = sy-subrc.
      ROLLBACK WORK.
      MESSAGE e999 WITH 'Error occurred when updating ZIM_IN_MAST'
                        'Return code ' w_subrc.
    ELSE.
      PERFORM update_in_mast_log USING w_action
                                       zim_in_mast.
    ENDIF.

    PERFORM create_reminder USING 'I'
                                  ' '
                                  '70'
                                  '01'
                                  ' '
                                  '00000000'.

    l_review_date = zim_screen-mdrp_date + 60.
    PERFORM create_reminder USING 'F'
                                  ' '
                                  '70'
                                  '04'
                                  ' '
                                  l_review_date.

  ENDLOOP.

  REFRESH r_cat.
  PERFORM add_r_cat USING '22'.
  PERFORM add_r_cat USING '27'.
  PERFORM correspondence USING zim_in_mast-pernr
                               zim_in_mast-incident
                               '            '
                               ' '.

* Unlock all the cases
  LOOP AT gt_outtab7 INTO gr_outtab7
       WHERE sel_code = abap_true.
    PERFORM unlock_im_pernr USING gr_outtab7-pernr.
  ENDLOOP.



ENDFORM.                    " SEND_TO_MDRP

*&      Form  SPLIT_EXTERNAL_AMOUNT
*--------------------------------------------------------------------*
*       text
*--------------------------------------------------------------------*
*      -->P_L_BEGDA  text
*      -->P_L_ENDDA  text
*--------------------------------------------------------------------*
FORM split_external_amount  USING    p_begda
                                     p_endda.

  DATA: l_amount             TYPE zim_in_rec_amt-amount,
         l_amount_day         TYPE zim_in_rec_amt-amount,
         l_amount_cum         TYPE zim_in_rec_amt-amount,
         l_days               TYPE i,
         l_start              TYPE d,
         l_end                TYPE d,
         l_initial            TYPE d,
         l_diff               TYPE i,
         l_mod                TYPE i,
         l_rem                TYPE i.

  l_amount = zim_in_rec_amt-ext_earnings.
  l_initial = '20090102'.   " First day of a payroll period

* Determine the amount for each working day
  l_start = p_begda.
  WHILE l_start LE p_endda.
    l_diff = l_start - l_initial.
    l_mod = l_diff MOD 7 .
    CASE l_mod.
      WHEN 0 OR 3 OR 4 OR 5 OR 6.
        ADD 1 TO l_days.
    ENDCASE.
    ADD 1 TO l_start.
  ENDWHILE.

  l_amount_day = l_amount / l_days.
  CLEAR l_amount_cum.

* Create a record for each week within the pay period
  l_start = p_begda.
  WHILE l_start LE p_endda.
    l_diff = l_start - l_initial.
    l_mod = l_diff MOD 7 .
    IF l_start = p_begda OR
       l_mod = 0.
      zim_in_rec_amt-begda = l_start.
      CLEAR zim_in_rec_amt-ext_earnings.
    ENDIF.
    CASE l_mod.
      WHEN 0 OR 3 OR 4 OR 5 OR 6.
        ADD l_amount_day TO zim_in_rec_amt-ext_earnings.
        ADD l_amount_day TO l_amount_cum.
    ENDCASE.

    IF l_start = p_endda OR
      l_mod = 6.
      zim_in_rec_amt-seq_no      = w_next_fix_seq_no.
      ADD 1 TO w_next_fix_seq_no.
      zim_in_rec_amt-endda = l_start.
      IF l_start = p_endda.
        IF l_amount_cum NE l_amount.
          zim_in_rec_amt-ext_earnings = zim_in_rec_amt-ext_earnings +
                                        l_amount - l_amount_cum.
        ENDIF.
      ENDIF.

      w_action                = 'I'.
      INSERT zim_in_rec_amt.
      IF sy-subrc NE 0.
        w_subrc = sy-subrc.
        ROLLBACK WORK.
        MESSAGE e999 WITH 'Error occurred when updating ZIM_IN_REC_AMT'
                          'Return code ' w_subrc.
      ELSE.
        PERFORM update_in_rec_amt_log USING w_action.
      ENDIF.
    ENDIF.
    ADD 1 TO l_start.
  ENDWHILE.

ENDFORM.                    " SPLIT_EXTERNAL_AMOUNT

  DATA: tbl_org        TYPE zhorgstruc OCCURS 0 WITH HEADER LINE,
         l_zlevel       TYPE zhorgstruc-zlevel.

  SELECT SINGLE * FROM zhorgstruc

           AND vbegda   <= sy-datum

  IF sy-subrc NE 0.

  l_zlevel = zhorgstruc-zlevel.

           INTO TABLE tbl_org
           WHERE lvl1_code = zim_screen-orgeh
             AND vbegda   <= sy-datum

           INTO TABLE tbl_org
           WHERE lvl3_code = zim_screen-orgeh
             AND vbegda   <= sy-datum
             AND vendda   >= sy-datum.
  ELSEIF l_zlevel = 4.
    SELECT * FROM zhorgstruc
           INTO TABLE tbl_org
           WHERE lvl4_code = zim_screen-orgeh
             AND vbegda   <= sy-datum
             AND vendda   >= sy-datum.
  ELSEIF l_zlevel = 5.
    SELECT * FROM zhorgstruc
           INTO TABLE tbl_org
           WHERE lvl5_code = zim_screen-orgeh
             AND vbegda   <= sy-datum
             AND vendda   >= sy-datum.
  ELSEIF l_zlevel = 6.

           INTO TABLE tbl_org
           WHERE lvl6_code = zim_screen-orgeh
             AND vbegda   <= sy-datum
             AND vendda   >= sy-datum.

           INTO TABLE tbl_org
           WHERE unit_code = zim_screen-orgeh
             AND vbegda   <= sy-datum
             AND vendda   >= sy-datum.

FORM delimit_leave .

  DATA: l_hours            TYPE pa0573-stdaz,
         l_last_day         TYPE d,
         l_date             TYPE d.

* The records should all be created for individual days.
  SELECT * FROM pa0573
         INTO TABLE tbl_0573
         WHERE pernr = zim_in_mast-pernr
           AND ( subty = '5500'
* Start CR 1694
*            OR   subty = '5501' )
             OR   subty = '5501'
             OR   subty = '5200' )
* End CR1694
           AND endda GE zim_in_cm-pid_date
           AND desc  = 'D&D CREATION OF 5501'.

  w_trans   = 'PA30'.
  w_mode    = 'N'.
  w_updt    = 'S'.
  w_sesnam  = 'DD_PID_DEL'.

* Delete records that commence on or after the PID date
  LOOP AT tbl_0573.
    IF tbl_0573-begda GE zim_in_cm-pid_date.
      PERFORM delete_5501_bdc.
      DELETE tbl_0573.
    ENDIF.
  ENDLOOP.

* program.  An employee may have two incidents, one may be PID
* but the other may still be open and have leave associated with
* it - i.e. uploaded from the roster.  This leave should NOT
* be deleted.

** Delimit records that commence before the PID date
** and finish on or after the PID date
*  LOOP AT tbl_0573.
*    REFRESH tbl_2001.
*    CLEAR l_hours.
*    IF tbl_0573-zpday = abap_true.             "part day absence
** read 2001 to find abs hours for each part day
*      SELECT * FROM pa2001

*&      Form  DETERMINE_STANDARD_NUMBR
*--------------------------------------------------------------------*
*       text
*--------------------------------------------------------------------*
*      -->P_W_S49_QCTYP  text
*      -->P_ZIM_SCREEN_S49_BEGDA  text
*      -->P_ZIM_SCREEN_S49_ENDDA  text
*      <--P_ZIM_SCREEN_S49_NUMBR  text
*--------------------------------------------------------------------*
FORM determine_standard_numbr  USING    p_pernr
                                        p_qctyp
                                        p_begda
                                        p_endda
                               CHANGING p_std_dur
                                        p_lve
                                        p_numbr.

  DATA: l_days               TYPE i,
         l_hours              TYPE pa0573-stdaz,
         l_rem                TYPE pa0573-stdaz,
         l_rem2               TYPE pa0573-stdaz,
         l_begda              TYPE d,
         l_endda              TYPE d,
         tb_t549q             TYPE t549q OCCURS 0 WITH HEADER LINE,
         l_std_hrs            TYPE pa0573-stdaz,
         l_std_days           TYPE i.

  CLEAR : p_std_dur, p_lve, p_numbr.
  REFRESH gt_outtab35_copy.
  CLEAR   gt_outtab35_copy.
  SELECT * FROM t549q
         INTO TABLE tb_t549q
         WHERE permo = '04'
           AND begda LE p_endda
           AND endda GE p_begda.
  LOOP AT tb_t549q.
    CLEAR gr_outtab35a.
    IF tb_t549q-begda LT p_begda.
      l_begda = p_begda.
    ELSE.
      l_begda = tb_t549q-begda.
    ENDIF.
    IF tb_t549q-endda GT p_endda.
      l_endda = p_endda.
    ELSE.
      l_endda = tb_t549q-endda.
    ENDIF.
    PERFORM get_standard_hours USING    p_pernr
                                        l_begda
                                        l_endda
                               CHANGING l_days
                                        l_hours.
    ADD l_hours TO l_std_hrs.
    ADD l_days  TO l_std_days.
    IF p_qctyp = 'WCDA'.
      gr_outtab35a-numbr = l_days.
      ADD l_days TO p_std_dur.
      PERFORM get_wc_leave_days  USING    p_pernr
                                          l_begda
                                          l_endda
                                 CHANGING gr_outtab35a-numbr_2.
    ELSE.
      gr_outtab35a-numbr = l_hours.
      ADD l_hours TO p_std_dur.
      PERFORM get_wc_leave_hours USING    p_pernr
                                          l_begda
                                          l_endda
                                 CHANGING gr_outtab35a-numbr_2.
    ENDIF.
    ADD gr_outtab35a-numbr_2 TO p_lve.
    gr_outtab35a-lve_begda = l_begda.
    gr_outtab35a-lve_endda = l_endda.
    APPEND gr_outtab35a TO gt_outtab35_copy.
  ENDLOOP.

  CLEAR zim_screen-s49_proc.
  LOOP AT gt_outtab35_copy INTO gr_outtab35a.
    LOOP AT gt_outtab35 INTO gr_outtab35
         WHERE lve_endda GE gr_outtab35a-lve_begda
           AND lve_begda LE gr_outtab35a-lve_endda.
      IF p_qctyp = 'WCDA'.
        ADD gr_outtab35-lve_days      TO   zim_screen-s49_proc.
        SUBTRACT gr_outtab35-lve_days FROM gr_outtab35a-numbr_2.
      ELSE.
        ADD gr_outtab35-lve_hrs      TO   zim_screen-s49_proc.
        SUBTRACT gr_outtab35-lve_hrs FROM gr_outtab35a-numbr_2.
      ENDIF.
    ENDLOOP.
    MODIFY gt_outtab35_copy FROM gr_outtab35a.
  ENDLOOP.

  l_rem = p_std_dur - zim_screen-s49_proc.
  l_rem2 = p_lve - zim_screen-s49_proc.
  IF l_rem LT 0.
    l_rem = 0.
  ENDIF.
  IF l_rem2 LT 0.
    l_rem2 = 0.
  ENDIF.
  IF l_rem GT l_rem2.
    l_rem = l_rem2.
  ENDIF.

  IF p_lve GT l_rem.
    p_numbr = l_rem.
  ELSE.
    p_numbr = p_lve.
  ENDIF.

ENDFORM.                    " DETERMINE_STANDARD_NUMBR

FORM delete_it0014  USING    p_pernr
                             p_incident
                             p_begda
                             p_endda.

  DATA: l_operation            TYPE  pspar-actio,
         l_dialog(1)            TYPE c,
         l_incident(10)         TYPE c,
         l_wclmn(15)            TYPE c,
         tb_pa0014              TYPE pa0014 OCCURS 0 WITH HEADER LINE.

  PERFORM lock_employee USING p_pernr
                              '0014'.

  IF w_error = abap_true.
    MESSAGE e999 WITH 'The employee is currently'
                      'being updated in PA30'
                      'by another user'.
  ENDIF.

* These functions need to be called to refresh the buffers.
  CALL FUNCTION 'HR_PSBUFFER_INITIALIZE'.
  CALL FUNCTION 'HRPAD00_FREE_PLOG_TAB'.
  CALL FUNCTION 'HR_PSA_CLEAR_BUFFER'.

  SELECT * FROM pa0014
         INTO TABLE tb_pa0014
         WHERE pernr = p_pernr
           AND ( subty = '9WCO'
            OR   subty = '9WCZ' )
           AND begda LE p_endda
           AND endda GE p_begda.

  LOOP AT tb_pa0014.
    SPLIT tb_pa0014-zuord AT ' ' INTO l_wclmn
                                      l_incident.
    SHIFT l_incident RIGHT DELETING TRAILING ' '.
    TRANSLATE l_incident USING ' 0'.
    IF l_incident NE p_incident.
      DELETE tb_pa0014.
    ENDIF.
  ENDLOOP.

  LOOP AT tb_pa0014.

    CLEAR p0014.

    IF tb_pa0014-begda GE p_begda AND
       tb_pa0014-endda LE p_endda.
      l_operation = 'DEL'.
      p0014-begda = tb_pa0014-begda.
      p0014-endda = tb_pa0014-endda.
    ELSE.
      l_operation = 'MOD'.
      IF tb_pa0014-begda GE p_begda.
        p0014-begda = p_endda + 1.
        p0014-endda = tb_pa0014-endda.
      ELSEIF tb_pa0014-endda LE p_endda.
        p0014-begda = tb_pa0014-begda.
        p0014-endda = p_begda - 1.
      ELSE.
        p0014-begda = tb_pa0014-begda.
        p0014-endda = p_begda - 1.
      ENDIF.
    ENDIF.

    p0014-pernr = p_pernr.
    p0014-infty = '0014'.
    p0014-subty = tb_pa0014-subty.
    p0014-lgart = tb_pa0014-lgart.
    p0014-waers = 'AUD'.
    p0014-anzhl = tb_pa0014-anzhl.
    p0014-betrg = tb_pa0014-betrg.
    p0014-zuord = tb_pa0014-zuord.
    p0014-preas = tb_pa0014-preas.
    l_dialog    = '1'.
*   l_dialog    = '2'.

    CALL FUNCTION 'HR_INFOTYPE_OPERATION'
      EXPORTING
        infty         = '0014'
        number        = tb_pa0014-pernr
        subtype       = tb_pa0014-subty
        lockindicator = tb_pa0014-sprps
        validityend   = tb_pa0014-endda
        validitybegin = tb_pa0014-begda
        recordnumber  = tb_pa0014-seqnr
        objectid      = tb_pa0014-objps
        record        = p0014
        operation     = l_operation
        dialog_mode   = l_dialog
      IMPORTING
        return        = w_bapiret
        key           = w_bapi_key.

    IF w_bapiret-type = 'E' OR
       w_bapiret-type = 'A'.
      w_error = abap_true.
      CLEAR w_ucomm.
      ROLLBACK WORK.
      MESSAGE ID     w_bapiret-id
              TYPE   w_bapiret-type
              NUMBER w_bapiret-number
              WITH   w_bapiret-message_v1
                     w_bapiret-message_v2
                     w_bapiret-message_v3
                     w_bapiret-message_v4.
    ENDIF.

* The original record spanned the period.  The first step would
* have delimited the record to before the period.  A new record needs
* to be created for after the period.
    IF tb_pa0014-begda LT p_begda AND
       tb_pa0014-endda GT p_endda.
      l_operation = 'INS'.
      p0014-begda = p_endda + 1.
      p0014-endda = tb_pa0014-endda.

      p0014-pernr = p_pernr.
      p0014-infty = '0014'.
      p0014-subty = tb_pa0014-subty.
      p0014-lgart = tb_pa0014-lgart.
      p0014-waers = 'AUD'.
      p0014-anzhl = tb_pa0014-anzhl.
      p0014-betrg = tb_pa0014-betrg.
      p0014-zuord = tb_pa0014-zuord.
      p0014-preas = tb_pa0014-preas.
      l_dialog    = '1'.
*     l_dialog    = '2'.
      CLEAR : p0014-seqnr, p0014-objps, p0014-sprps.

      CALL FUNCTION 'HR_INFOTYPE_OPERATION'
        EXPORTING
          infty         = '0014'
          number        = tb_pa0014-pernr
          subtype       = tb_pa0014-subty
          lockindicator = p0014-sprps
          validityend   = p0014-endda
          validitybegin = p0014-begda
          recordnumber  = p0014-seqnr
          objectid      = p0014-objps
          record        = p0014
          operation     = l_operation
          dialog_mode   = l_dialog
        IMPORTING
          return        = w_bapiret
          key           = w_bapi_key.

      IF w_bapiret-type = 'E' OR
         w_bapiret-type = 'A'.
        w_error = abap_true.
        CLEAR w_ucomm.
        ROLLBACK WORK.
        MESSAGE ID     w_bapiret-id
                TYPE   w_bapiret-type
                NUMBER w_bapiret-number
                WITH   w_bapiret-message_v1
                       w_bapiret-message_v2
                       w_bapiret-message_v3
                       w_bapiret-message_v4.
      ENDIF.
    ENDIF.

  ENDLOOP.

  PERFORM unlock_employee USING p_pernr
                                '0014'.

         INTO TABLE tbl_zim_in_cert
         WHERE pernr    = zim_in_mast-pernr

    ENDIF.

       tbl_zim_in_cert-status = 'P'.
      l_start = abap_true.
      l_begda = tbl_zim_in_cert-date_from.
    ENDIF.

                                zim_in_mast-incident

*&      Form  GET_WC_LEAVE_HOURS
*--------------------------------------------------------------------*
*       text
*--------------------------------------------------------------------*
*      -->P_P_PERNR  text
*      -->P_P_BEGDA  text
*      -->P_P_ENDDA  text
*      <--P_P_LVE  text
*--------------------------------------------------------------------*
FORM get_wc_leave_hours  USING    p_pernr
                                  p_begda
                                  p_endda
                         CHANGING p_lve.

  DATA: l_endda            TYPE d,
         l_begda2           TYPE d,
         l_endda2           TYPE d,
         tb_pa2001          TYPE pa2001 OCCURS 0 WITH HEADER LINE.

  l_endda = p_endda + 1.
  CLEAR p_lve.

  LOOP AT gt_outtab38 INTO gr_outtab38.
    IF gr_outtab38-endda LT p_begda OR
       gr_outtab38-begda GT l_endda.
      CONTINUE.
    ENDIF.

    IF gr_outtab38-endda = p_begda AND
       gr_outtab38-vtken = abap_true.         " Previous day flag
      CONTINUE.
    ENDIF.

    IF gr_outtab38-begda = l_endda AND
       gr_outtab38-vtken = ' '.         " Previous day flag
      CONTINUE.
    ENDIF.

    IF gr_outtab38-zpday = abap_true.
      SELECT * FROM pa2001
             INTO TABLE tb_pa2001
             WHERE pernr = p_pernr
               AND begda GE p_begda
               AND endda LE l_endda
               AND beguz =  gr_outtab38-beguz
               AND enduz =  gr_outtab38-enduz
               AND vtken =  gr_outtab38-vtken.

      LOOP AT tb_pa2001.
        IF tb_pa2001-endda = p_begda AND
           tb_pa2001-vtken = abap_true.         " Previous day flag
          CONTINUE.
        ENDIF.

        IF tb_pa2001-begda = l_endda AND
           tb_pa2001-vtken = ' '.         " Previous day flag
          CONTINUE.
        ENDIF.

        ADD tb_pa2001-stdaz TO p_lve.
      ENDLOOP.
    ELSE.
* Full day leave - we don't need to worry about previous day flag
      IF gr_outtab38-begda LT p_begda.
        l_begda2 = p_begda.
      ELSE.
        l_begda2 = gr_outtab38-begda.
      ENDIF.
      IF gr_outtab38-endda GT p_endda.
        l_endda2 = p_endda.
      ELSE.
        l_endda2 = gr_outtab38-endda.
      ENDIF.
      PERFORM read_infotype USING p_pernr l_begda2 l_endda2.

      CALL FUNCTION 'HR_PERSONAL_WORK_SCHEDULE'
        EXPORTING
          pernr         = p_pernr
          begda         = l_begda2
          endda         = l_endda2
        TABLES
          i0001         = p0001
          i0007         = p0007
          perws         = pws
        EXCEPTIONS
          error_occured = 1
          abort_occured = 2
          OTHERS        = 3.

      CHECK sy-subrc = 0.
      LOOP AT pws
           WHERE stdaz > 0
             AND datum GE l_begda2
             AND datum LE l_endda2.
        IF pws-tprog <> 'FREE' AND
           ( pws-tprog = '****' OR pws-ftkla = '0' ).
          ADD pws-stdaz TO p_lve.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.




ENDFORM.                    " GET_WC_LEAVE_HOURS

  DATA: l_endda            TYPE d,
         l_begda2           TYPE d,
         l_endda2           TYPE d,
         l_date             TYPE d,
         tb_pa2001          TYPE pa2001 OCCURS 0 WITH HEADER LINE.

  DATA: BEGIN OF tb_days  OCCURS 0,
            date         TYPE d,
            day(1)       TYPE c,
         END OF tb_days.

  REFRESH tb_days.
  CLEAR tb_days.
  l_date = p_begda.
  WHILE l_date LE p_endda.
    tb_days-date = l_date.
    APPEND tb_days.
    ADD 1 TO l_date.
  ENDWHILE.

  l_endda = p_endda + 1.
  CLEAR p_lve.

  LOOP AT gt_outtab38 INTO gr_outtab38.
    IF gr_outtab38-endda LT p_begda OR
       gr_outtab38-begda GT l_endda.
      CONTINUE.
    ENDIF.

    IF gr_outtab38-endda = p_begda AND
       gr_outtab38-vtken = abap_true.         " Previous day flag
      CONTINUE.
    ENDIF.

    IF gr_outtab38-begda = l_endda AND
       gr_outtab38-vtken = ' '.         " Previous day flag
      CONTINUE.
    ENDIF.

    IF gr_outtab38-zpday = abap_true.
      SELECT * FROM pa2001
             INTO TABLE tb_pa2001
             WHERE pernr = p_pernr
               AND begda GE gr_outtab38-begda
               AND endda LE gr_outtab38-endda
               AND beguz =  gr_outtab38-beguz
               AND enduz =  gr_outtab38-enduz
               AND vtken =  gr_outtab38-vtken.

      LOOP AT tb_pa2001.
        IF tb_pa2001-endda = p_begda AND
           tb_pa2001-vtken = abap_true.         " Previous day flag
          CONTINUE.
        ENDIF.

        IF tb_pa2001-begda = l_endda AND
           tb_pa2001-vtken = ' '.         " Previous day flag
          CONTINUE.
        ENDIF.

        IF tb_pa2001-stdaz GT 0.
          LOOP AT tb_days
               WHERE date = tb_pa2001-begda.
            tb_days-day  = abap_true.
            MODIFY tb_days.
          ENDLOOP.
        ENDIF.
      ENDLOOP.
    ELSE.
* Full day leave - we don't need to worry about previous day flag
      IF gr_outtab38-begda LT p_begda.
        l_begda2 = p_begda.
      ELSE.
        l_begda2 = gr_outtab38-begda.
      ENDIF.
      IF gr_outtab38-endda GT p_endda.
        l_endda2 = p_endda.
      ELSE.
        l_endda2 = gr_outtab38-endda.
      ENDIF.
      PERFORM read_infotype USING p_pernr l_begda2 l_endda2.
      PERFORM read_infotype USING p_pernr l_begda2 l_endda2.

      CALL FUNCTION 'HR_PERSONAL_WORK_SCHEDULE'
        EXPORTING
          pernr         = p_pernr
          begda         = l_begda2
          endda         = l_endda2
        TABLES
          i0001         = p0001
          i0007         = p0007
          perws         = pws
        EXCEPTIONS
          error_occured = 1
          abort_occured = 2
          OTHERS        = 3.

      CHECK sy-subrc = 0.
      LOOP AT pws
           WHERE stdaz > 0
             AND datum GE l_begda2
             AND datum LE l_endda2.
        IF pws-tprog <> 'FREE' AND
           ( pws-tprog = '****' OR pws-ftkla = '0' ).
          IF pws-stdaz GT 0.
            LOOP AT tb_days
                 WHERE date = pws-datum.
              tb_days-day  = abap_true.
              MODIFY tb_days.
            ENDLOOP.
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

  LOOP AT tb_days
       WHERE day = abap_true.
    ADD 1 TO p_lve.
  ENDLOOP.


ENDFORM.                    " GET_WC_LEAVE_DAYS

FORM calculate_ip_dates .
  DATA: lv_dis_date     TYPE datum,
        lv_ip_begda     TYPE datum,
        lv_ip_endda     TYPE datum.

  PERFORM get_ip_records.
* disablement date
  lv_dis_date = gr_ip_rec->get_disablement_date( ).
  IF lv_dis_date IS NOT INITIAL
    AND lv_dis_date NE zim_in_cm-disablement_date.

  gr_ip_rec->get_ip_dates( IMPORTING ev_begda = lv_ip_begda

  IF lv_ip_begda NE zim_in_cm-ip_start_date

    zim_in_cm-ip_start_date = lv_ip_begda.
  ENDIF.

     lv_ip_endda IS NOT INITIAL.
    zim_in_cm-ip_end_date = lv_ip_endda.
  ENDIF.

  IF lv_dis_date IS NOT INITIAL AND
     lv_ip_begda NE zim_in_cm-day_271.
    zim_in_cm-day_271 = lv_ip_begda.
  ELSE.
    zim_in_cm-day_271 = zim_in_cm-ip_start_date.
  ENDIF.

FORM save_primary_claim .
  DATA: lt_prim_claim     TYPE TABLE OF zim_in_prim,
        ls_prim_claim     TYPE zim_in_prim,
        lt_date_range     TYPE TABLE OF zca_date_range_upd,
        lt_prim_claim_upd TYPE TABLE OF zim_in_prim_upd.

  FIELD-SYMBOLS:
        <ls_date_range>   TYPE zca_date_range_upd.

* if date from exists, update the data in db
  CHECK zim_in_prim-begda IS NOT INITIAL AND
        gr_prim_claim     IS  NOT INITIAL.

* get records from db for the date range
  CALL METHOD gr_prim_claim->get_primary_claims_for_date
    EXPORTING
      iv_begda      = zim_in_prim-begda
      iv_endda      = zim_in_prim-endda
    IMPORTING
      et_prim_claim = lt_prim_claim.

* form the date range table to pass it to zcl_olims_prim_claim=>delimit_date_ranges
  LOOP AT lt_prim_claim  INTO ls_prim_claim.
    APPEND INITIAL LINE TO lt_date_range ASSIGNING <ls_date_range>.
    <ls_date_range>-source_idx = sy-tabix.
    <ls_date_range>-begda = ls_prim_claim-begda.
    <ls_date_range>-endda = ls_prim_claim-endda.
  ENDLOOP.

* delimit or overwrite records
  CALL METHOD zcl_olims_prim_claim=>add_date_range
    EXPORTING
      iv_begda      = zim_in_prim-begda
      iv_endda      = zim_in_prim-endda
      iv_overlap    = w_unsworn_claim       "Overlap allowed for unsworn employees
    CHANGING
      ct_date_range = lt_date_range.

* Convert
  CALL METHOD gr_prim_claim->convert_date_range_to_prim_upd
    EXPORTING
      it_date_range     = lt_date_range
      it_prim_table     = lt_prim_claim
      is_prim_new       = zim_in_prim
    IMPORTING
      et_prim_claim_upd = lt_prim_claim_upd.

* Put update table to Object
  CALL METHOD gr_prim_claim->put_prim_upd
    EXPORTING
      it_prim_claim_upd = lt_prim_claim_upd.

* Save changes to DB
  CALL METHOD gr_prim_claim->save_prim_claim_to_db( ).

* Determine if any medical certificates exist for the period
* that could result in changed WC Recoup Benefit records
  PERFORM check_med_cert USING 'M'
                               zim_in_prim-begda
                               zim_in_prim-endda.

* Refresh ALV data
  CLEAR gr_prim_claim.    "clear out the object
  w_3415_refresh = abap_true.
  CLEAR zim_in_prim.

*&---------------------------------------------------------------------*
*&      Form  GET_CURRENT_CERT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_GR_OUTTAB7_PERNR  text
*      -->P_GR_OUTTAB7_INCIDENT  text
*----------------------------------------------------------------------*
FORM get_current_cert  USING    p_pernr
                                p_incident
                       CHANGING p_status
                                p_type
                                p_cap_hours
                                p_cap_days
                                p_cap_hours2
                                p_cap_days2.

  DATA: tbl_in_cert  LIKE zim_in_cert     OCCURS 0 WITH HEADER LINE.

  SELECT * FROM zim_in_cert
         INTO TABLE tbl_in_cert
         WHERE pernr    = p_pernr
           AND incident = p_incident
           AND date_from LE sy-datum
           AND inactive  = ' '.
  SELECT * FROM zim_in_cap
         APPENDING CORRESPONDING FIELDS OF TABLE tbl_in_cert
         WHERE pernr      = p_pernr
           AND incident   = p_incident
           AND date_from LE sy-datum
           AND inactive   = ' '.
  SORT tbl_in_cert BY mc_date DESCENDING
                      date_from DESCENDING
                      date_to DESCENDING.

* Cater for situations where a PID record exists without an end date
* but it has been superseded by a Suitable Duties record
  LOOP AT tbl_in_cert.
    IF tbl_in_cert-date_to GE sy-datum OR
       tbl_in_cert-date_to IS INITIAL.
      p_status    = tbl_in_cert-status.
      p_cap_hours = tbl_in_cert-cap_hours1.
      p_cap_days  = tbl_in_cert-cap_days1.
      p_cap_hours2 = tbl_in_cert-cap_hours2.
      p_cap_days2 = tbl_in_cert-cap_days2.

      SELECT SINGLE * FROM zim_in_cap
             WHERE pernr    = tbl_in_cert-pernr
               AND incident = tbl_in_cert-incident
               AND seq_no   = tbl_in_cert-seq_no.
      IF sy-subrc NE 0.
        p_type = 'M'.
      ENDIF.
    ENDIF.
    EXIT.
  ENDLOOP.


ENDFORM.                    " GET_CURRENT_CERT

FORM update_injuries .

* The injuries have changed.
* Delete the current injuries and re-add the new injuries

  DATA : l_inc_inj            TYPE zim_inc_inj,
         l_inc_inj_log        TYPE zim_inc_inj_log,
         l_seq                TYPE zp902_seq_no,
         tb_inc_inj           TYPE STANDARD TABLE OF zim_inc_inj.

  SELECT * FROM zim_inc_inj
         INTO TABLE tb_inc_inj
         WHERE pernr   = p9001-pernr
           AND identnr = p9001-identnr.

    DELETE zim_inc_inj FROM l_inc_inj.

    IF sy-subrc NE 0.
      w_subrc = sy-subrc.
      ROLLBACK WORK.

    ELSE.

                                       l_inc_inj.
    ENDIF.

    ADD 1 TO l_seq.
    l_inc_inj-seq_no  = l_seq.
    l_inc_inj-pernr   = p9001-pernr.
    l_inc_inj-identnr = p9001-identnr.
    INSERT zim_inc_inj FROM l_inc_inj.

    IF sy-subrc NE 0.
      w_subrc = sy-subrc.
      ROLLBACK WORK.
      MESSAGE e999 WITH 'Error occurred when updating ZIM_INC_INJ'
                        'Return code ' w_subrc.
    ELSE.
      PERFORM update_inc_inj_log USING 'I'
                                       l_inc_inj.
    ENDIF.

  ENDLOOP.



ENDFORM.                    " UPDATE_INJURIES
