  METHOD  handle_user_double_click.
*--------------------------------------------------------------------*
    w_node_12 = node_key.
    PERFORM get_employee  USING    sy-uname
                          CHANGING w_user_pernr
                                   w_user_name.

    IF sy-tcode = 'ZIM_MGR'
    OR sy-tcode = 'ZIM_FTCTL'
    OR sy-tcode = 'ZIM_MGRDEP'
    OR sy-tcode = 'ZIM_DEP'.
      w_rpt_restrict = abap_true.

*   PERFORM read_tbl_menu.

      WHEN 'TABLE_A'.
        CALL FUNCTION 'Z_OLIMS_HEALTH_OVERVIEW'.
      WHEN 'TABLE_B'.
        CALL FUNCTION 'Z_OLIMS_PROVIDER_OVERVIEW'.
